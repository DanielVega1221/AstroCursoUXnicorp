    <section class="mb-16">
      <h2>Tipos Básicos</h2>
      
      <div class="bg-gradient-to-r from-green-50 to-emerald-50 dark:from-green-900/20 dark:to-emerald-900/20 p-6 rounded-lg mb-6">
        <h3 class="text-lg font-semibold mb-4">🏷️ ¿Qué es un "tipo"?</h3>
        
        <p class="mb-4">
          Un <strong>tipo</strong> es como una etiqueta que le pones a una variable para decir <strong>qué clase de información guarda</strong>.
        </p>
        
        <div class="grid md:grid-cols-3 gap-4">
          <div class="bg-white dark:bg-gray-800 p-4 rounded">
            <p class="font-semibold text-blue-600 mb-2">📝 string</p>
            <p class="text-sm">Texto - palabras, frases</p>
            <code class="text-xs">"Hola", "Ana", "123"</code>
          </div>
          
          <div class="bg-white dark:bg-gray-800 p-4 rounded">
            <p class="font-semibold text-green-600 mb-2">🔢 number</p>
            <p class="text-sm">Números - enteros, decimales</p>
            <code class="text-xs">42, 3.14, -10</code>
          </div>
          
          <div class="bg-white dark:bg-gray-800 p-4 rounded">
            <p class="font-semibold text-purple-600 mb-2">✅ boolean</p>
            <p class="text-sm">Verdadero o Falso</p>
            <code class="text-xs">true, false</code>
          </div>
        </div>
      </div>
      
      <p class="mb-4">
        En TypeScript, declaras el tipo de una variable con <strong>dos puntos</strong> <code>:</code> después del nombre:
      </p>
      
      <CodeBlock
        language="typescript"
        code={`// Tipos primitivos
const nombre: string = "Ana";
const edad: number = 25;
const activo: boolean = true;

// Arrays
const numeros: number[] = [1, 2, 3];
const palabras: Array<string> = ["hola", "mundo"];

// Objetos
interface Usuario {
  nombre: string;
  edad: number;
  email?: string; // opcional
}

const usuario: Usuario = {
  nombre: "Ana",
  edad: 25
};`}
      />
      
      <div class="bg-gray-50 dark:bg-gray-800 p-6 rounded-lg mt-6">
        <h4 class="text-lg font-semibold mb-4">🔍 Explicación línea por línea:</h4>
        
        <div class="space-y-4">
          <div class="border-l-4 border-blue-500 pl-4">
            <code class="text-sm font-semibold">const nombre: string = "Ana";</code>
            <p class="text-sm mt-2">
              <strong>nombre</strong> es una variable que SOLO puede guardar texto (string). 
              Si intentas poner un número, TypeScript te detiene.
            </p>
          </div>
          
          <div class="border-l-4 border-green-500 pl-4">
            <code class="text-sm font-semibold">const numeros: number[] = [1, 2, 3];</code>
            <p class="text-sm mt-2">
              <strong>numeros</strong> es un array (lista) que SOLO puede contener números. 
              Los corchetes <code>[]</code> significan "array de".
            </p>
            <p class="text-xs mt-1 text-gray-600">
              También puedes escribirlo como: <code>Array&lt;number&gt;</code> - es lo mismo, dos formas de decir lo mismo.
            </p>
          </div>
          
          <div class="border-l-4 border-purple-500 pl-4">
            <code class="text-sm font-semibold">interface Usuario {'{'} ... {'}'}</code>
            <p class="text-sm mt-2">
              Una <strong>interface</strong> es como un molde o plantilla. Define la forma que debe tener un objeto.
            </p>
            <p class="text-xs mt-1 text-gray-600">
              Piensa en ello como un formulario: "Todo Usuario debe tener nombre (texto), edad (número), y opcionalmente email".
            </p>
          </div>
          
          <div class="border-l-4 border-yellow-500 pl-4">
            <code class="text-sm font-semibold">email?: string;</code>
            <p class="text-sm mt-2">
              El signo <code>?</code> significa <strong>opcional</strong>. Este campo puede existir o no. 
              Sin el <code>?</code>, TypeScript te obligaría a incluirlo siempre.
            </p>
          </div>
        </div>
      </div>
      
      <div class="bg-purple-50 dark:bg-purple-900/20 p-4 rounded mt-6">
        <p class="font-semibold mb-2">💡 ¿Por qué usar tipos?</p>
        <p class="text-sm mb-3">
          Imagina que tienes una función que suma dos números. Sin tipos, alguien podría pasarle texto por accidente:
        </p>
        <div class="grid md:grid-cols-2 gap-4 text-sm">
          <div class="bg-red-100 dark:bg-red-900/30 p-3 rounded">
            <p class="font-semibold text-red-700 dark:text-red-300 mb-1">Sin TypeScript:</p>
            <code>sumar("5", "3")</code>
            <p class="mt-1">Resultado: "53" 🤦 (unió texto)</p>
          </div>
          <div class="bg-green-100 dark:bg-green-900/30 p-3 rounded">
            <p class="font-semibold text-green-700 dark:text-green-300 mb-1">Con TypeScript:</p>
            <code>sumar("5", "3")</code>
            <p class="mt-1">❌ ERROR: "Los parámetros deben ser números"</p>
          </div>
        </div>
      </div>
    </section>

      <h2>Props Tipadas en React</h2>
      
      <div class="bg-gradient-to-r from-orange-50 to-red-50 dark:from-orange-900/20 dark:to-red-900/20 p-6 rounded-lg mb-6">
        <h3 class="text-lg font-semibold mb-4">🎁 ¿Qué son las Props?</h3>
        
        <p class="mb-4">
          <strong>Props</strong> (abreviatura de "properties" = propiedades) son como los <strong>argumentos o parámetros</strong> que le pasas a un componente de React.
        </p>
        
        <div class="bg-white dark:bg-gray-800 p-4 rounded mb-4">
          <p class="font-semibold mb-2">🎯 Analogía:</p>
          <p class="text-sm mb-3">
            Imagina que un componente es una <strong>máquina de hacer sándwiches</strong>. Las props son los ingredientes que le das:
          </p>
          <ul class="text-sm space-y-1">
            <li>Le das: pan, jamón, queso (props)</li>
            <li>La máquina: hace el sándwich (componente)</li>
            <li>Obtienes: un sándwich de jamón y queso (resultado)</li>
          </ul>
        </div>
        
        <p class="mb-4">
          Sin TypeScript, la máquina aceptaría cualquier cosa (¿arena? ¿zapatos?). 
          Con TypeScript, defines exactamente qué ingredientes acepta.
        </p>
      </div>
      
      <h3>Ejemplo: Botón con Props Tipadas</h3>
      
      <p class="mb-4">
        Vamos a crear un componente de botón que acepta configuración específica:
      </p>
      
      <CodeBlock
        language="typescript"
        filename="Button.tsx"
        code={`interface ButtonProps {
  text: string;
  onClick: () => void;
  variant?: 'primary' | 'secondary';
}

export default function Button({ text, onClick, variant = 'primary' }: ButtonProps) {
  return (
    <button onClick={onClick} className={\`btn btn-\${variant}\`}>
      {text}
    </button>
  );
}`}
      />
      
      <div class="bg-gray-50 dark:bg-gray-800 p-6 rounded-lg mt-6">
        <h4 class="text-lg font-semibold mb-4">🔍 Desglosando el código:</h4>
        
        <div class="space-y-4">
          <div class="bg-white dark:bg-gray-700 p-4 rounded">
            <code class="text-sm font-semibold text-blue-600">interface ButtonProps {'{'}</code>
            <p class="text-sm mt-2">
              Definimos la "receta" de ingredientes que acepta nuestro botón. Es como escribir en la máquina: 
              "Solo acepto estos ingredientes específicos".
            </p>
          </div>
          
          <div class="bg-white dark:bg-gray-700 p-4 rounded">
            <code class="text-sm font-semibold text-green-600">text: string;</code>
            <p class="text-sm mt-2">
              El botón DEBE recibir un texto (string). Sin texto, no hay botón. Es obligatorio.
            </p>
            <p class="text-xs text-gray-600 mt-1">Ejemplo: <code>text="Enviar"</code></p>
          </div>
          
          <div class="bg-white dark:bg-gray-700 p-4 rounded">
            <code class="text-sm font-semibold text-purple-600">onClick: () =&gt; void;</code>
            <p class="text-sm mt-2">
              Una <strong>función</strong> que se ejecuta cuando haces clic. <code>() =&gt; void</code> significa: 
              "una función que no recibe parámetros y no devuelve nada".
            </p>
            <p class="text-xs text-gray-600 mt-1">Ejemplo: <code>onClick={'{'}{'}'}() =&gt; console.log('Click!'){'}'}{'}'}</code></p>
          </div>
          
          <div class="bg-white dark:bg-gray-700 p-4 rounded">
            <code class="text-sm font-semibold text-orange-600">variant?: 'primary' | 'secondary';</code>
            <p class="text-sm mt-2 mb-2">
              El <code>?</code> lo hace <strong>opcional</strong>. El <code>|</code> significa "o". 
              Solo puede ser "primary" o "secondary", nada más.
            </p>
            <div class="bg-yellow-50 dark:bg-yellow-900/30 p-2 rounded text-xs">
              <p class="font-semibold mb-1">Esto se llama "Union Type":</p>
              <p>Es como decir: "puede ser rojo O azul, pero nunca verde"</p>
            </div>
          </div>
          
          <div class="bg-white dark:bg-gray-700 p-4 rounded">
            <code class="text-sm font-semibold text-pink-600">variant = 'primary'</code>
            <p class="text-sm mt-2">
              Valor por defecto. Si no le pasas variant, usa 'primary' automáticamente. 
              Es como un ingrediente que ya viene incluido si no especificas otro.
            </p>
          </div>
        </div>
      </div>
      
      <h3 class="mt-8">Usando el Botón</h3>
      
      <p class="mb-4">Así es como usarías este botón en tu código:</p>
      
      <CodeBlock

// ❌ ERROR - falta onClick (es obligatorio)

// ❌ ERROR - variant solo puede ser 'primary' o 'secondary'

      />

          <li>✅ <strong>Refactorización segura:</strong> Si cambias una prop, TypeScript te avisa dónde más debes actualizar</li>

      <h3>1. Type Guards (Guardias de Tipo)</h3>
      
      <p class="mb-4">
        Los <strong>Type Guards</strong> son formas de decirle a TypeScript: "confía en mí, sé qué tipo es esto". 
        TypeScript es inteligente y puede <strong>estrechar</strong> (narrow) los tipos automáticamente.
      </p>
      
      <div class="bg-blue-50 dark:bg-blue-900/20 p-4 rounded-lg mb-4">
        <p class="font-semibold mb-2">🎯 El problema:</p>
        <p class="text-sm">
          Tienes una variable que puede ser <code>string | number</code>. ¿Cómo trabajas con ella sin errores?
        </p>
      </div>
      
      <CodeBlock
        language="typescript"
        code={`// ✅ typeof - para tipos primitivos
function procesar(valor: string | number) {
  if (typeof valor === "string") {
    // TypeScript SABE que aquí valor es string
    return valor.toUpperCase();
  }
  // TypeScript SABE que aquí valor es number
  return valor * 2;
}

// ✅ instanceof - para clases
function manejarError(error: Error | string) {
  if (error instanceof Error) {
    console.log(error.message);
  } else {
    console.log(error);
  }
}

// ✅ in - para propiedades de objetos
type Perro = { nombre: string; ladrar: () => void };
type Gato = { nombre: string; maullar: () => void };

function hacerSonido(animal: Perro | Gato) {
  if ("ladrar" in animal) {
    animal.ladrar(); // TypeScript sabe que es Perro
  } else {
    animal.maullar(); // TypeScript sabe que es Gato
  }
}

// 🚀 Custom Type Guard - para casos complejos
function esNumero(valor: unknown): valor is number {
  return typeof valor === "number";
}

function multiplicar(a: unknown, b: unknown) {
  if (esNumero(a) && esNumero(b)) {
    return a * b; // TypeScript confía en tu guardia
  }
  throw new Error("Los valores deben ser números");
}`}
      />
      
      <div class="bg-gray-50 dark:bg-gray-800 p-4 rounded-lg mt-4">
        <h4 class="font-semibold mb-2">🔍 ¿Qué es <code>valor is number</code>?</h4>
        <p class="text-sm mb-2">
          Es un <strong>Type Predicate</strong> (predicado de tipo). Le dices a TypeScript: 
          "si esta función retorna true, entonces el valor ES definitivamente un number".
        </p>
        <p class="text-sm text-gray-600 dark:text-gray-400">
          Es como darle un certificado: "yo verifiqué esto, puedes confiar".
        </p>
      </div>

      <h3 class="mt-8">2. Type Assertions y "as const"</h3>
      
      <p class="mb-4">
        A veces <strong>tú sabes más que TypeScript</strong> sobre qué tipo es algo. Puedes "forzarlo" con <code>as</code>.
      </p>
      
      <CodeBlock
        language="typescript"
        code={`// Type Assertion - "yo sé mejor que tú, TypeScript"
const miElemento = document.getElementById("app") as HTMLDivElement;
miElemento.style.color = "red"; // Sin "as", TypeScript pensaría que puede ser null

// as const - hacer valores INMUTABLES y LITERALES
const RUTAS = {
  home: "/",
  about: "/about",
  contact: "/contact"
} as const;

// Sin "as const": tipo = { home: string; about: string; ... }
// CON "as const": tipo = { readonly home: "/"; readonly about: "/about"; ... }

type Ruta = typeof RUTAS[keyof typeof RUTAS]; // "/" | "/about" | "/contact"

// ✅ Útil para configuraciones que nunca cambian
const COLORES = ["rojo", "azul", "verde"] as const;
type Color = typeof COLORES[number]; // "rojo" | "azul" | "verde"

// ❌ Sin as const, sería: type Color = string (muy genérico)`}
      />
      
      <div class="bg-yellow-50 dark:bg-yellow-900/20 p-4 rounded-lg mt-4">
        <p class="font-semibold mb-2">⚠️ Cuidado con Type Assertions</p>
        <p class="text-sm">
          Usar <code>as</code> es como decirle a TypeScript "confía en mí". Si te equivocas, 
          TypeScript no te protegerá y tendrás errores en runtime. <strong>Úsalo solo cuando estés seguro.</strong>
        </p>
      </div>

      <h3 class="mt-8">3. Utility Types Avanzados</h3>
      
      <p class="mb-4">
        TypeScript incluye tipos "helper" que transforman otros tipos. Son súper útiles para evitar repetición.
      </p>
      
      <CodeBlock
        language="typescript"
        code={`interface Usuario {
  id: number;
  nombre: string;
  email: string;
  password: string;
  createdAt: Date;
}

// 🔹 Partial - todas las propiedades opcionales
type ActualizarUsuario = Partial<Usuario>;
// = { id?: number; nombre?: string; email?: string; ... }

// 🔹 Pick - seleccionar solo algunas propiedades
type UsuarioPublico = Pick<Usuario, "id" | "nombre">;
// = { id: number; nombre: string; }

// 🔹 Omit - excluir propiedades
type UsuarioSinPassword = Omit<Usuario, "password">;
// = { id: number; nombre: string; email: string; createdAt: Date; }

// 🔹 Record - crear objeto con claves y valores específicos
type RolePermisos = Record<"admin" | "user" | "guest", string[]>;
// = { admin: string[]; user: string[]; guest: string[]; }

const permisos: RolePermisos = {
  admin: ["leer", "escribir", "eliminar"],
  user: ["leer", "escribir"],
  guest: ["leer"]
};

// 🔹 ReturnType - obtener el tipo de retorno de una función
function obtenerUsuario() {
  return { id: 1, nombre: "Ana" };
}

type Usuario2 = ReturnType<typeof obtenerUsuario>;
// = { id: number; nombre: string; }

// 🔹 Awaited - obtener el tipo de una Promise
type RespuestaAPI = Awaited<Promise<{ data: string }>>;
// = { data: string; }`}
      />
      
      <div class="bg-green-50 dark:bg-green-900/20 p-4 rounded-lg mt-4">
        <p class="font-semibold mb-2">💡 Caso de uso real</p>
        <p class="text-sm mb-3">
          Imagina una API que devuelve un Usuario completo, pero tu formulario solo necesita actualizar nombre y email:
        </p>
        <code class="text-xs bg-white dark:bg-gray-800 p-2 rounded block">
          type FormularioUsuario = Pick&lt;Usuario, "nombre" | "email"&gt;;
        </code>
        <p class="text-sm mt-2">
          Si después agregas más campos a Usuario, no tienes que tocar FormularioUsuario. ¡Reutilización!
        </p>
      </div>

      <h3 class="mt-8">4. Mapped Types (Tipos Mapeados)</h3>
      
      <p class="mb-4">
        Los <strong>Mapped Types</strong> te permiten crear nuevos tipos transformando las propiedades de un tipo existente. 
        Es como hacer un <code>map()</code> pero sobre tipos.
      </p>
      
      <CodeBlock
        language="typescript"
        code={`interface Producto {
  id: number;
  nombre: string;
  precio: number;
}

// Hacer todas las propiedades string (para formularios)
type ProductoFormulario = {
  [K in keyof Producto]: string;
};
// = { id: string; nombre: string; precio: string; }

// Hacer todas las propiedades opcionales y null
type ProductoParcial = {
  [K in keyof Producto]?: Producto[K] | null;
};

// 🚀 Ejemplo avanzado: agregar "loading" a cada campo
type ConLoading<T> = {
  [K in keyof T]: {
    value: T[K];
    loading: boolean;
  };
};

type ProductoConLoading = ConLoading<Producto>;
// = {
//   id: { value: number; loading: boolean; };
//   nombre: { value: string; loading: boolean; };
//   precio: { value: number; loading: boolean; };
// }

// ✅ Útil para estados de carga en formularios complejos
const producto: ProductoConLoading = {
  id: { value: 1, loading: false },
  nombre: { value: "Laptop", loading: false },
  precio: { value: 1200, loading: true }
};`}
      />

      <h3 class="mt-8">5. Conditional Types (Tipos Condicionales)</h3>
      
      <p class="mb-4">
        Los tipos pueden tener lógica condicional: "si es A, entonces B, sino C". 
        Es como un <code>if</code> pero para tipos.
      </p>
      
      <CodeBlock

  description?: string;

type CardProps = BaseCardProps & (

  | { variant?: "default" | "minimal"; image?: string }
);

function isFeatured(props: CardProps): props is CardProps & { variant: "featured" } {
  return props.variant === "featured";
}

// 5. Mapped type para estilos

  [K in CardSize]: {
    padding: string;
    fontSize: string;
  };
};

const styles: StyleMap = {
  sm: { padding: "0.5rem", fontSize: "0.875rem" },
  md: { padding: "1rem", fontSize: "1rem" },
  lg: { padding: "1.5rem", fontSize: "1.25rem" }
};

  const { title, description, size = "md", variant = "default" } = props;
