<section class="mb-16">
  <h2>Sintaxis Básica</h2>

  <h3>Variables</h3>
  
  <p>Las variables <strong>almacenan datos</strong> que puedes usar y modificar. Usa <code>const</code> por defecto (no cambia), <code>let</code> si necesitas reasignar, y evita <code>var</code> (obsoleto).</p>
  
  <div class="card bg-yellow-50 dark:bg-yellow-900/20 mb-4">
    <div class="text-sm space-y-2">
      <p><strong>🟢 const</strong> - No se puede reasignar (usa esto por defecto)</p>
      <p><strong>🟡 let</strong> - Se puede reasignar (para contadores, estados)</p>
      <p><strong>🔴 var</strong> - Antiguo, tiene problemas de scope. NO USAR.</p>
    </div>
  </div>
  
  <CodeBlock
    language="javascript"
    filename="variables.js"
    code={`// const: no se puede reasignar (recomendado por defecto)
const nombre = "Ana";
const edad = 25;

// let: se puede reasignar
let contador = 0;
contador = contador + 1;

// var: antiguo, no usar
var antiguaVariable = "no usar";

// Tipos de datos
const texto = "Hola";           // String
const numero = 42;               // Number
const decimal = 3.14;            // Number
const verdadero = true;          // Boolean
const falso = false;             // Boolean
const nulo = null;               // Null
const indefinido = undefined;    // Undefined`}
  />

  <p>Los operadores realizan operaciones entre valores. Existen <strong>aritméticos</strong> (matemáticos), de <strong>comparación</strong> (devuelven true/false), y <strong>lógicos</strong> (combinan condiciones).</p>
  
  <div class="grid md:grid-cols-3 gap-4 mb-4">

      <code class="text-xs">=== !== &lt; &gt;</code>
    </div>
    <div class="card bg-purple-50 dark:bg-purple-900/20">
      <h4 class="text-sm font-semibold mb-2">Lógicos</h4>

<section class="mb-16">

const frutas = ["manzana", "banana", "naranja"];
const numeros = [1, 2, 3, 4, 5];

// Acceder a elementos (índice empieza en 0)
console.log(frutas[0]);  // "manzana"
console.log(frutas[2]);  // "naranja"

// Longitud
console.log(frutas.length);  // 3

// Agregar elementos
frutas.push("uva");          // Añade al final
frutas.unshift("fresa");     // Añade al inicio

// Eliminar elementos
frutas.pop();                // Elimina el último
frutas.shift();              // Elimina el primero

// Métodos útiles
const numeros = [1, 2, 3, 4, 5];

// map: transforma cada elemento
const dobles = numeros.map(num => num * 2);
// [2, 4, 6, 8, 10]

// filter: filtra elementos
const pares = numeros.filter(num => num % 2 === 0);
// [2, 4]

// find: encuentra el primer elemento que cumple
const primerPar = numeros.find(num => num % 2 === 0);
// 2

// reduce: reduce a un solo valor
const suma = numeros.reduce((total, num) => total + num, 0);
// 15

// forEach: ejecuta función por cada elemento
numeros.forEach(num => {
  console.log(num);
});

// some: verifica si al menos uno cumple
const hayPares = numeros.some(num => num % 2 === 0);
// true

// every: verifica si todos cumplen
const todosPares = numeros.every(num => num % 2 === 0);
// false`}

    language="javascript"
    filename="objetos.js"
    code={`// Crear objeto
const persona = {
  nombre: "Ana",
  edad: 25,
  ciudad: "Madrid",
  programadora: true
};

// Acceder a propiedades

console.log(persona["edad"]);    // 25

// Modificar propiedades
persona.edad = 26;

// Agregar propiedades
persona.email = "ana@ejemplo.com";

// Métodos (funciones dentro de objetos)

  sumar: (a, b) => a + b,
  restar: (a, b) => a - b
};

console.log(calculadora.sumar(5, 3));  // 8

console.log(nombre);  // "Ana"

// Spread operator
const personaCompleta = {
  ...persona,
  pais: "España"
};

// Object.keys, Object.values, Object.entries
const persona = { nombre: "Ana", edad: 25 };

Object.keys(persona);     // ["nombre", "edad"]
Object.values(persona);   // ["Ana", 25]
Object.entries(persona);  // [["nombre", "Ana"], ["edad", 25]]`}

  <CodeBlock
    language="javascript"
    filename="async.js"
    code={`// Promise tradicional
fetch('https://api.ejemplo.com/datos')
  .then(response => response.json())
  .then(data => {
    console.log(data);
  })
  .catch(error => {
    console.error('Error:', error);
  });

// Async/Await (más limpio)
async function obtenerDatos() {
  try {
    const response = await fetch('https://api.ejemplo.com/datos');
    const data = await response.json();
    console.log(data);
  } catch (error) {
    console.error('Error:', error);
  }
}

obtenerDatos();

// Múltiples requests en paralelo
async function obtenerMultiple() {
  const [usuarios, posts] = await Promise.all([
    fetch('/api/usuarios').then(r => r.json()),
    fetch('/api/posts').then(r => r.json())
  ]);
  
  console.log(usuarios, posts);
}`}

async function obtenerUsuarios() {

async function crearPost(datos) {
  const response = await fetch('https://jsonplaceholder.typicode.com/posts', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json'
    },
    body: JSON.stringify(datos)
  });

  return resultado;
}
